# 方案6: SFU架构 实施计划

**Goal:** 支持10-20人房间，超过6人自动切换到SFU模式。

**Architecture:** 扩展信令服务器添加UDP转发功能，客户端根据人数自动切换P2P/SFU。

```c
static SOCKET g_udp_sock = INVALID_SOCKET;

/* 初始化UDP转发socket */
static int init_udp_relay(uint16_t port) {
    g_udp_sock = socket(AF_INET, SOCK_DGRAM, 0);
    if (g_udp_sock == INVALID_SOCKET) return 0;

    struct sockaddr_in local;
    local.sin_family = AF_INET;
    local.sin_addr.s_addr = INADDR_ANY;
    local.sin_port = htons(port);

    if (bind(g_udp_sock, (struct sockaddr*)&local, sizeof(local)) == SOCKET_ERROR) {
        closesocket(g_udp_sock);
        g_udp_sock = INVALID_SOCKET;
        return 0;
    }
    return 1;
}
```

- [ ] **Step 2: 添加SFU转发线程**

```c
/* SFU转发：收到UDP包后转发给同房间其他成员 */
static DWORD WINAPI sfu_relay_thread(LPVOID arg) {
    uint8_t buf[1500];
    struct sockaddr_in from;
    int from_len;

    while (1) {
        from_len = sizeof(from);
        int n = recvfrom(g_udp_sock, (char*)buf, sizeof(buf), 0,
                         (struct sockaddr*)&from, &from_len);
        if (n <= 0) continue;

        /* 查找发送者所属房间 */
        char room[MAX_NAME] = {0};
        EnterCriticalSection(&g_lock);
        for (int i = 0; i < MAX_CLIENTS; i++) {
            if (g_clients[i].active &&
                g_clients[i].addr.sin_addr.s_addr == from.sin_addr.s_addr &&
                g_clients[i].local_port == ntohs(from.sin_port)) {
                strncpy(room, g_clients[i].room, MAX_NAME - 1);
                break;
            }
        }
        LeaveCriticalSection(&g_lock);

        if (room[0] == 0) continue;  /* Unknown sender */

        /* 转发给同房间其他成员 */
        EnterCriticalSection(&g_lock);
        for (int i = 0; i < MAX_CLIENTS; i++) {
            if (g_clients[i].active && strcmp(g_clients[i].room, room) == 0 &&
                !(g_clients[i].addr.sin_addr.s_addr == from.sin_addr.s_addr &&
                  g_clients[i].local_port == ntohs(from.sin_port))) {
                struct sockaddr_in dest;
                dest.sin_family = AF_INET;
                dest.sin_addr = g_clients[i].addr;
                dest.sin_port = htons(g_clients[i].local_port);
                sendto(g_udp_sock, (const char*)buf, n, 0,
                       (struct sockaddr*)&dest, sizeof(dest));
            }
        }
        LeaveCriticalSection(&g_lock);
    }
    return 0;
}
```

```c
#ifndef SFU_CLIENT_H
#define SFU_CLIENT_H

#include <stdint.h>
#include <winsock2.h>

typedef struct {
    SOCKET           udp_sock;
    struct sockaddr_in server_addr;
    int              connected;
    char             local_id[32];

    void             (*recv_cb)(const char *peer_id, const uint8_t *data, int len, void *user);

void sfu_close(sfu_client_t *sfu);
int  sfu_send(sfu_client_t *sfu, const uint8_t *data, int len);
void sfu_set_local_id(sfu_client_t *sfu, const char *id);

#endif
```

- [ ] **Step 2: 实现SFU客户端**

```c
// src/sfu_client.c
#include "sfu_client.h"
#include <string.h>
#include <stdio.h>

#pragma comment(lib, "ws2_32.lib")

static DWORD WINAPI sfu_recv_thread(LPVOID arg) {
    sfu_client_t *sfu = (sfu_client_t*)arg;
    uint8_t buf[1500];
    struct sockaddr_in from;
    int from_len;

    while (sfu->connected) {
        from_len = sizeof(from);
        int n = recvfrom(sfu->udp_sock, (char*)buf, sizeof(buf), 0,
                         (struct sockaddr*)&from, &from_len);
        if (n <= 0) continue;

        /* 检查是否来自服务器 */
        if (from.sin_addr.s_addr == sfu->server_addr.sin_addr.s_addr &&
            from.sin_port == sfu->server_addr.sin_port) {
            /* 解析包头获取发送者ID */
            if (n < 32) continue;
            char peer_id[33] = {0};
            memcpy(peer_id, buf, 32);

            if (sfu->recv_cb)
                sfu->recv_cb(peer_id, buf + 32, n - 32, sfu->user_data);
        }
    }
    return 0;
}

int sfu_init(sfu_client_t *sfu, const char *server_ip, uint16_t server_port,
             void (*cb)(const char*, const uint8_t*, int, void*), void *user) {
    memset(sfu, 0, sizeof(*sfu));
    sfu->recv_cb = cb;
    sfu->user_data = user;

    sfu->udp_sock = socket(AF_INET, SOCK_DGRAM, 0);
    if (sfu->udp_sock == INVALID_SOCKET) return 0;

    sfu->server_addr.sin_family = AF_INET;
    sfu->server_addr.sin_port = htons(server_port);
    inet_pton(AF_INET, server_ip, &sfu->server_addr.sin_addr);

    sfu->connected = 1;
    CreateThread(NULL, 0, sfu_recv_thread, sfu, 0, NULL);

    return 1;
}

void sfu_close(sfu_client_t *sfu) {
    sfu->connected = 0;
    if (sfu->udp_sock != INVALID_SOCKET) {
        closesocket(sfu->udp_sock);
        sfu->udp_sock = INVALID_SOCKET;
    }
}

int sfu_send(sfu_client_t *sfu, const uint8_t *data, int len) {
    uint8_t buf[1500];
    /* 添加32字节头（本地ID） */
    memset(buf, 0, 32);
    memcpy(buf, sfu->local_id, strlen(sfu->local_id));
    memcpy(buf + 32, data, len);

    return sendto(sfu->udp_sock, (const char*)buf, 32 + len, 0,
                  (struct sockaddr*)&sfu->server_addr, sizeof(sfu->server_addr)) > 0;
}

void sfu_set_local_id(sfu_client_t *sfu, const char *id) {
    strncpy(sfu->local_id, id, sizeof(sfu->local_id) - 1);
    sfu->local_id[sizeof(sfu->local_id) - 1] = 0;
}
```
