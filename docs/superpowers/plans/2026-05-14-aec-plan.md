```c
// src/aec.h
#ifndef AEC_H
#define AEC_H

#include <stdint.h>

/* AEC configuration */
#define AEC_SAMPLE_RATE    16000
#define AEC_FRAME_SIZE     160    /* 10ms @ 16kHz */
#define AEC_TAIL_LENGTH    1024   /* 64ms tail */

typedef struct aec_t aec_t;

/* Create AEC instance */
aec_t* aec_create(void);

/* Destroy AEC instance */
void aec_destroy(aec_t *aec);

/* Process one frame of near-end (microphone) signal
 * near_in:  input microphone signal (160 samples)
 * far_ref:  reference signal from speaker (160 samples)
 * near_out: output cleaned signal (160 samples)
 *
 * The AEC estimates the echo path and subtracts the estimated echo
 * from the near-end signal.
 */
void aec_process(aec_t *aec, const short *near_in, const short *far_ref, short *near_out);

/* Reset AEC state (e.g., when delay changes significantly) */
void aec_reset(aec_t *aec);

#endif
```

- [ ] **Step 4: 实现简单AEC**

```c
// src/aec.c
#include "aec.h"
#include <stdlib.h>
#include <string.h>
#include <math.h>

/* Simple AEC using NLMS (Normalized Least Mean Squares) adaptive filter */
struct aec_t {
    float filter[AEC_TAIL_LENGTH];     /* Adaptive filter coefficients */
    float far_buffer[AEC_TAIL_LENGTH]; /* Far-end reference buffer */
    int   far_write_pos;               /* Write position in far buffer */
    float power;                       /* Running power estimate */
    float step_size;                   /* Adaptation step size */
};

aec_t* aec_create(void) {
    aec_t *aec = (aec_t*)calloc(1, sizeof(aec_t));
    if (!aec) return NULL;

    aec->step_size = 0.5f;  /* Initial step size */
    aec->power = 1.0f;

    return aec;
}

void aec_destroy(aec_t *aec) {
    free(aec);
}

void aec_process(aec_t *aec, const short *near_in, const short *far_ref, short *near_out) {
    /* Store far-end reference in circular buffer */
    for (int i = 0; i < AEC_FRAME_SIZE; i++) {
        aec->far_buffer[aec->far_write_pos] = (float)far_ref[i];
        aec->far_write_pos = (aec->far_write_pos + 1) % AEC_TAIL_LENGTH;
    }

    /* Process each sample */
    for (int i = 0; i < AEC_FRAME_SIZE; i++) {
        /* Calculate echo estimate using adaptive filter */
        float echo_estimate = 0.0f;
        float power = 0.0f;

        for (int j = 0; j < AEC_TAIL_LENGTH; j++) {
            int idx = (aec->far_write_pos - AEC_FRAME_SIZE + i - j + AEC_TAIL_LENGTH) % AEC_TAIL_LENGTH;
            float far_sample = aec->far_buffer[idx];
            echo_estimate += aec->filter[j] * far_sample;
            power += far_sample * far_sample;
        }

        /* Normalize power */
        power /= AEC_TAIL_LENGTH;
        aec->power = 0.9f * aec->power + 0.1f * power;
        if (aec->power < 1.0f) aec->power = 1.0f;

        /* Calculate error (near - echo_estimate) */
        float near_sample = (float)near_in[i];
        float error = near_sample - echo_estimate;

        /* Update filter coefficients (NLMS) */
        float step = aec->step_size / (aec->power + 1e-10f);
        for (int j = 0; j < AEC_TAIL_LENGTH; j++) {
            int idx = (aec->far_write_pos - AEC_FRAME_SIZE + i - j + AEC_TAIL_LENGTH) % AEC_TAIL_LENGTH;
            aec->filter[j] += step * error * aec->far_buffer[idx];
        }

        /* Output the error signal (cleaned) */
        if (error > 32767.0f) error = 32767.0f;
        if (error < -32768.0f) error = -32768.0f;
        near_out[i] = (short)error;
    }
}

void aec_reset(aec_t *aec) {
    memset(aec->filter, 0, sizeof(aec->filter));
    memset(aec->far_buffer, 0, sizeof(aec->far_buffer));
    aec->far_write_pos = 0;
    aec->power = 1.0f;
}
```

修改 `wavein_cb` 回调：
```c
#include "aec.h"

extern aec_t *g_aec;  /* From main.c */

/* Buffer for AEC reference signal */
static short g_aec_ref_buf[160] = {0};
static int   g_aec_ref_pos = 0;

/* Called from audio_playback to provide reference signal */
void audio_capture_set_far_ref(const short *samples, int frames) {
    for (int i = 0; i < frames && g_aec_ref_pos < 160; i++) {
        g_aec_ref_buf[g_aec_ref_pos++] = samples[i];
    }
}

static void CALLBACK wavein_cb(HWAVEIN hwi, UINT msg, DWORD_PTR inst, DWORD_PTR param1, DWORD_PTR param2) {
    (void)hwi; (void)param2;
    if (msg != WIM_DATA) return;
    audio_capture_t *ac = (audio_capture_t*)inst;
    if (!ac || !ac->running || !ac->callback) return;

    WAVEHDR *hdr = (WAVEHDR*)param1;
    int frames = hdr->dwBytesRecorded / sizeof(short);
    const short *input = (const short*)hdr->lpData;

    if (g_aec && frames == 160) {
        /* Apply AEC */
        short cleaned[160];
        aec_process(g_aec, input, g_aec_ref_buf, cleaned);
        ac->callback(cleaned, frames, ac->user_data);

        /* Reset reference buffer */
        g_aec_ref_pos = 0;
    } else {
        /* No AEC or wrong frame size, pass through */
        ac->callback(input, frames, ac->user_data);
    }

    /* Re-queue the buffer */
    waveInAddBuffer(g_wavein, hdr, sizeof(WAVEHDR));
}
```

```c
extern void audio_capture_set_far_ref(const short *samples, int frames);

void audio_playback_submit(audio_playback_t *ap, const short *samples, int frames) {
    /* Provide reference to AEC */
    audio_capture_set_far_ref(samples, frames);

    /* Original ring buffer code */
    for (int i = 0; i < frames; i++) {
        LONG head = ap->rb_head;
        LONG tail = ap->rb_tail;
        if (head - tail >= RB_MASK + 1) break;
        ap->rb_data[head & RB_MASK] = samples[i];
        MemoryBarrier();
        ap->rb_head = head + 1;
    }
}
```
